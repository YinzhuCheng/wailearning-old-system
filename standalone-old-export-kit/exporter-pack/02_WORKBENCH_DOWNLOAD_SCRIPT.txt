#!/usr/bin/env bash
set -Eeuo pipefail

###############################################################################
# 旧系统独立导出：Workbench 下载脚本版
# 用法：
# 1. 在阿里云 Workbench 用 root 登录旧系统服务器。
# 2. 通常不需要改任何内容；如路径特殊，只改下面几个变量。
# 3. 整段复制到 Workbench，按回车即可从公开 GitHub 下载脚本并导出。
# 4. 导出完成后，还要把迁移包下载到 Windows；Windows 本地校验和 .sha256 哈希校验都是可选检查。
###############################################################################

export APP_NAME="wailearning-old-system"
export ENV_FILE="/opt/wailearning-old-system/current/.env.production"
export RELEASE_DIR="/opt/wailearning-old-system/current"
export OUT_DIR="/root/wailearning-migration"
export EXTRA_UPLOAD_DIRS=""

export KIT_OWNER="YinzhuCheng"
export KIT_REPO="wailearning-old-system"
export KIT_BRANCH="main"
export KIT_SCRIPT="standalone-old-export-kit/scripts/standalone_old_export.sh"
export KIT_VERIFY_SCRIPT="standalone-old-export-kit/scripts/verify_export_bundle.sh"

if [ "$(id -u)" -ne 0 ]; then
  echo "请用 root 用户运行。Workbench 请确认已经 root 登录。"
  exit 1
fi

install_curl() {
  if command -v curl >/dev/null 2>&1; then
    return 0
  fi
  echo "当前系统缺少 curl，先安装下载工具。"
  if command -v apt-get >/dev/null 2>&1; then
    apt-get update
    DEBIAN_FRONTEND=noninteractive apt-get install -y curl ca-certificates
  elif command -v dnf >/dev/null 2>&1; then
    dnf install -y curl ca-certificates
  elif command -v yum >/dev/null 2>&1; then
    yum install -y curl ca-certificates
  else
    echo "未识别包管理器，无法自动安装 curl。"
    exit 1
  fi
}

download_file() {
  local url="$1"
  local output="$2"
  curl -fsSL --retry 3 --connect-timeout 20 "$url" -o "$output"
}

RUN_DIR="/root/wailearning-migration-kit"
REMOTE_SCRIPT="${RUN_DIR}/standalone_old_export.sh"
REMOTE_VERIFY_SCRIPT="${RUN_DIR}/verify_export_bundle.sh"
RAW_BASE="https://raw.githubusercontent.com"
SCRIPT_URL="${RAW_BASE}/${KIT_OWNER}/${KIT_REPO}/${KIT_BRANCH}/${KIT_SCRIPT}"
VERIFY_SCRIPT_URL="${RAW_BASE}/${KIT_OWNER}/${KIT_REPO}/${KIT_BRANCH}/${KIT_VERIFY_SCRIPT}"

mkdir -p "${RUN_DIR}" "${OUT_DIR}"
install_curl

echo "下载独立导出脚本：${SCRIPT_URL}"
download_file "${SCRIPT_URL}" "${REMOTE_SCRIPT}"
chmod +x "${REMOTE_SCRIPT}"

echo "下载导出包校验脚本：${VERIFY_SCRIPT_URL}"
download_file "${VERIFY_SCRIPT_URL}" "${REMOTE_VERIFY_SCRIPT}"
chmod +x "${REMOTE_VERIFY_SCRIPT}"

echo "开始导出旧系统迁移包。"
bash "${REMOTE_SCRIPT}"

LATEST_BUNDLE="$(ls -1t "${OUT_DIR}"/old-to-new-*.tar.gz 2>/dev/null | head -n 1 || true)"
if [ -n "${LATEST_BUNDLE}" ]; then
  echo
  echo "建议立即校验导出包："
  echo "bash ${REMOTE_VERIFY_SCRIPT} ${LATEST_BUNDLE}"
  echo
  echo "然后把迁移包下载到 Windows："
  echo "1. ${LATEST_BUNDLE}"
  echo "可选：${LATEST_BUNDLE}.sha256（用于哈希校验）"
  echo
  echo "可选：Windows 下载后，可用 05_windows_verify_bundle.ps1 校验。"
fi
