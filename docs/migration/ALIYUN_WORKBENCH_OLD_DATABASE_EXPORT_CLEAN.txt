#!/usr/bin/env bash
set -Eeuo pipefail

###############################################################################
# 旧数据库导出操作流：阿里云 Workbench 干净版
# 使用方法：
# 1. 默认假设你已经 root 登录旧系统服务器。
# 2. 通常只需要确认 ENV_FILE / RELEASE_DIR 是否符合你的部署路径。
# 3. 整段复制到 Workbench，按回车即可导出迁移包。
#
# 不需要记住数据库密码；脚本会从 .env.production 读取 DATABASE_URL。
# 导出内容包含数据库、上传附件、旧环境文件、systemd/nginx/git 快照和清单。
###############################################################################

export APP_NAME="wailearning-old-system"
export ENV_FILE="/opt/wailearning-old-system/current/.env.production"
export RELEASE_DIR="/opt/wailearning-old-system/current"
export MIGRATION_OUTPUT_DIR="/root/wailearning-migration"
export MIGRATION_LABEL="old-to-new-$(date +%Y%m%d%H%M%S)"

export GITHUB_OWNER="YinzhuCheng"
export GITHUB_REPO="wailearning-old-system"
export GITHUB_BRANCH="main"
export RAW_BASE="https://raw.githubusercontent.com"
export EXPORT_SCRIPT_PATH="docs/migration/export_migration_bundle.sh"

###############################################################################
# 以下内容通常不需要修改。
###############################################################################

if [ "$(id -u)" -ne 0 ]; then
  echo "请用 root 用户运行。Workbench 请确认已经 root 登录。"
  exit 1
fi

assert_path_exists() {
  local name="$1"
  local value="$2"
  if [ -z "${value}" ] || [ ! -e "${value}" ]; then
    echo "路径不存在，请检查 ${name}：${value}"
    exit 1
  fi
}

ensure_downloader() {
  if command -v curl >/dev/null 2>&1; then
    return 0
  fi
  echo "当前系统缺少 curl，先安装下载工具。"
  if command -v apt-get >/dev/null 2>&1; then
    apt-get update
    DEBIAN_FRONTEND=noninteractive apt-get install -y curl ca-certificates
  elif command -v dnf >/dev/null 2>&1; then
    dnf install -y curl ca-certificates
  elif command -v yum >/dev/null 2>&1; then
    yum install -y curl ca-certificates
  else
    echo "未识别系统包管理器，无法自动安装 curl。"
    exit 1
  fi
}

download_file() {
  local url="$1"
  local output="$2"
  curl -fsSL --retry 3 --connect-timeout 20 "$url" -o "$output"
}

assert_path_exists "ENV_FILE" "${ENV_FILE}"
assert_path_exists "RELEASE_DIR" "${RELEASE_DIR}"

RUN_DIR="/root/${APP_NAME}-migration"
REMOTE_SCRIPT="${RUN_DIR}/export_migration_bundle.sh"
mkdir -p "${RUN_DIR}" "${MIGRATION_OUTPUT_DIR}"

ensure_downloader
EXPORT_SCRIPT_URL="${RAW_BASE}/${GITHUB_OWNER}/${GITHUB_REPO}"
EXPORT_SCRIPT_URL="${EXPORT_SCRIPT_URL}/${GITHUB_BRANCH}"
EXPORT_SCRIPT_URL="${EXPORT_SCRIPT_URL}/${EXPORT_SCRIPT_PATH}"
echo "下载仓库内最新迁移导出脚本：${EXPORT_SCRIPT_URL}"
download_file "${EXPORT_SCRIPT_URL}" "${REMOTE_SCRIPT}"
chmod +x "${REMOTE_SCRIPT}"

echo "开始导出迁移包。"
echo "输出目录：${MIGRATION_OUTPUT_DIR}"
bash "${REMOTE_SCRIPT}"
